% ---------------- Q7 ----------------
\item Show that the complex exponential signal $x(t) = \ejw{\omega t}$ is periodic and that its fundamental period is $2\pi/\omega$.

\begin{keyresult}
$T_0 = 2\pi/\omega$ is the smallest positive $T$ satisfying $\ejw{\omega(t+T)} = \ejw{\omega t}$.
\end{keyresult}
